cmake_minimum_required(VERSION 3.22)
project(BadlionMenuUI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(badlionmenuui SHARED src/BadlionMenu.cpp src/AndroidAutoHost.cpp)
target_include_directories(badlionmenuui PRIVATE include sdk_compat src)
target_compile_options(badlionmenuui PRIVATE -Wall -Wextra -Werror=return-type -fno-exceptions -fno-rtti -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_options(badlionmenuui PRIVATE -Wl,--gc-sections)
target_link_libraries(badlionmenuui PRIVATE log dl)
if(NOT PRELOADER_SO)
 message(FATAL_ERROR "Provide PRELOADER_SO: path to matching LeviLaunchroid 1.6.1 libpreloader.so")
endif()
add_library(preloader SHARED IMPORTED)
set_target_properties(preloader PROPERTIES IMPORTED_LOCATION "${PRELOADER_SO}")
target_link_libraries(badlionmenuui PRIVATE preloader)
set_target_properties(badlionmenuui PROPERTIES OUTPUT_NAME badlionmenuui)
